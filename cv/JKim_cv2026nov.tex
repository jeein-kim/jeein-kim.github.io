%% --- keep a heading with what follows (needspace.sty is not installed here) -
%% Measures what is left on the page and starts a new one outright. A penalty-
%% based version invites a break just before the heading, which makes longtable
%% flush its header above the heading and then repeat it on the next page.
\newcommand{\cvneedspace}[1]{%
  \dimen0=#1\relax
  \ifdim\pagegoal=\maxdimen\else
    \ifdim\dimen0>\dimexpr\pagegoal-\pagetotal\relax\newpage\fi
  \fi}

%% --- section heading: small-caps title over a thin rule --------------------
\newcommand{\cvsection}[2][]{%
  \cvneedspace{6\baselineskip}%  keep a heading with the first rows of its table
  \vspace{1.9ex}%
  \noindent{\color{darkgreen}\fontsize{13}{15.6}\selectfont\bfseries
    \hspace*{0.45em}\raisebox{0.05em}{\rotatebox[origin=c]{45}{\rule{0.5em}{0.5em}}}\hspace{0.3em}
    {\addfontfeature{LetterSpace=3}\MakeUppercase{#2}}%
    \hfill{\normalfont\small\strut #1}}\par\vspace{0.5ex}%
  {\color{darkgreen}\hrule height 1pt}\par\vspace{1.1ex}%
  \nopagebreak}

%% --- grey note set flush right, sharing the line when it fits ---------------
%% Falls to its own right-aligned line when it does not, instead of breaking
%% mid-phrase the way a bare \hfill does.
\newcommand{\sidenote}[1]{%
  \unskip\nobreak\hfill\penalty50\hskip1em\hbox{}\nobreak\hfill
  \mbox{\footnotesize\color{notegrey}#1}}
